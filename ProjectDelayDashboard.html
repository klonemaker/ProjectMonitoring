        let globalData = [];

        let currentZone = "All";

        let currentPeriod = "All";

        // Helper function to extract data from a row regardless of exact header capitalization/spacing

        function getCol(rowObj, partialName) {

            const key = Object.keys(rowObj).find(k => k.toLowerCase().includes(partialName.toLowerCase()));

            return key ? rowObj[key] : "";

        }


        // Format dates safely

        function formatVal(v) {

            if (v instanceof Date) return v.getFullYear().toString();

            return (v || "").toString().trim();

        }


        document.getElementById('fileUpload').addEventListener('change', function(e) {

            const file = e.target.files[0];

            if (!file) return;


            document.getElementById('loadingStatus').textContent = 'Extracting and cleaning data from sheets...';


            const reader = new FileReader();

            reader.onload = function(event) {

                try {

                    const data = new Uint8Array(event.target.result);

                    const workbook = XLSX.read(data, {type: 'array', cellDates: true});

                    globalData = [];


                    // Loop through all sheets and extract specific Active sheets

                    workbook.SheetNames.forEach(sheetName => {

                        let period = "";

                        const lowerName = sheetName.toLowerCase();

                        

                        if (lowerName.includes("before 2014")) period = "Before 2014";

                        else if (lowerName.includes("14-19")) period = "2014-19";

                        else if (lowerName.includes("19-24")) period = "2019-24";

                        else return; // Skip non-relevant sheets


                        const worksheet = workbook.Sheets[sheetName];

                        const rawData = XLSX.utils.sheet_to_json(worksheet, {header: 1, defval: "", raw: true});

                        

                        let headerRowIndex = -1;

                        let headers = [];


                        // Find the header row by looking for key PMO PPT columns

                        for (let i = 0; i < rawData.length; i++) {

                            const rowStr = rawData[i].map(v => (v||"").toString().trim().toLowerCase());

                            if (rowStr.includes("rly") && rowStr.includes("name of project") && rowStr.includes("time overrun reason")) {

                                headerRowIndex = i;

                                headers = rawData[i].map(h => (h||"").toString().trim());

                                break;

                            }

                        }


                        if (headerRowIndex !== -1) {

                            for (let i = headerRowIndex + 1; i < rawData.length; i++) {

                                const rowArr = rawData[i];

                                let rowObj = {};

                                for (let j = 0; j < headers.length; j++) {

                                    if (headers[j]) rowObj[headers[j]] = rowArr[j];

                                }


                                const zone = getCol(rowObj, 'rly');

                                const reason = getCol(rowObj, 'time overrun reason');

                                const projName = getCol(rowObj, 'name of project');

                                

                                if (zone && reason && projName) {

                                    globalData.push({

                                        'Period': period,

                                        'Zone': formatVal(zone),

                                        'Reason': formatVal(reason),

                                        'Project Name': formatVal(projName),

                                        'Sanctioned Year': formatVal(getCol(rowObj, 'sanctioned year')),

                                        'Length': formatVal(getCol(rowObj, 'length of project')),

                                        'Orig Cost': formatVal(getCol(rowObj, 'original sanctioned cost')),

                                        'Antic Cost': formatVal(getCol(rowObj, 'anticipated cost'))

                                    });

                                }

                            }

                        }

                    });


                    if (globalData.length === 0) {

                        alert("Could not find the expected sheets (before 2014, 14-19, 19-24) or columns.");

                        document.getElementById('loadingStatus').textContent = 'No data extracted.';

                        return;

                    }


                    document.getElementById('uploadSection').style.display = 'none';

                    document.getElementById('dashboard').style.display = 'block';

                    

                    renderSelectors();

                    applyFilters();


                } catch (error) {

                    document.getElementById('loadingStatus').textContent = 'Error processing file.';

                    console.error(error);

                }

            };

            reader.readAsArrayBuffer(file);

        });


        function renderSelectors() {

            const zones = new Set();

            const periods = new Set();

            const reasons = new Set();


            globalData.forEach(row => {

                if (row['Zone']) zones.add(row['Zone']);

                if (row['Period']) periods.add(row['Period']);

                if (row['Reason']) reasons.add(row['Reason']);

            });


            // Build Zone Grid

            let zHtml = `<div class="zone-btn active" onclick="setFilter('zone', 'All', this)">All</div>`;

            Array.from(zones).sort().forEach(z => {

                zHtml += `<div class="zone-btn" onclick="setFilter('zone', '${z}', this)">${z}</div>`;

            });

            document.getElementById('zoneContainer').innerHTML = zHtml;


            // Build Period Pills

            let pHtml = `<div class="class-pill active" onclick="setFilter('period', 'All', this)">All Periods</div>`;

            Array.from(periods).sort().forEach(p => {

                pHtml += `<div class="class-pill" onclick="setFilter('period', '${p}', this)">${p}</div>`;

            });

            document.getElementById('periodContainer').innerHTML = pHtml;


            // Build Reason Pills

            let rHtml = `<div class="class-pill active" onclick="setFilter('reason', 'All', this)">All Reasons</div>`;

            Array.from(reasons).sort().forEach(r => {

                rHtml += `<div class="class-pill" onclick="setFilter('reason', '${r}', this)">${r}</div>`;

            });

            document.getElementById('reasonContainer').innerHTML = rHtml;

        }


        function setFilter(type, val, element) {

            if (type === 'zone') {

                currentZone = val;

                document.querySelectorAll('#zoneContainer .zone-btn').forEach(el => el.classList.remove('active'));

            } else if (type === 'period') {

                currentPeriod = val;

                document.querySelectorAll('#periodContainer .class-pill').forEach(el => el.classList.remove('active'));

            } else if (type === 'reason') {

                currentReason = val;

                document.querySelectorAll('#reasonContainer .class-pill').forEach(el => el.classList.remove('active'));

            }

            element.classList.add('active');

            applyFilters();

        }


        function applyFilters() {

            document.getElementById('selectAll').checked = false;

            updateBulkUI();

            

            const searchInput = document.getElementById('searchBar').value.trim();

            const searchText = searchInput.toLowerCase();


            const filteredData = globalData.filter(row => {

                const projName = row['Project Name'].toLowerCase();

                const matchSearch = searchText === "" || projName.includes(searchText);

                const matchZone = currentZone === "All" || row['Zone'] === currentZone;

                const matchPeriod = currentPeriod === "All" || row['Period'] === currentPeriod;

                const matchReason = currentReason === "All" || row['Reason'] === currentReason;


                return matchSearch && matchZone && matchPeriod && matchReason;

            });


            renderTable(filteredData, searchInput);

        }


        function renderTable(data, searchInput) {

            const tbody = document.getElementById('tableBody');

            document.getElementById('recordCount').textContent = `Showing ${data.length} projects`;

            

            tbody.innerHTML = "";


            if (data.length === 0) {

                tbody.innerHTML = '<tr><td colspan="10" style="text-align: center; padding: 20px;">No matching projects found.</td></tr>';

                return;

            }


            const fragment = document.createDocumentFragment();


            data.forEach((row, idx) => {

                const tr = document.createElement('tr');

                tr.id = `row-${idx}`;

                let projHtml = row['Project Name'];

                

                const shareText = `🚧 *Project Delay Alert*\n\n*Project:* ${row['Project Name']}\n*Zone:* ${row['Zone']} | *Period:* ${row['Period']}\n*Reason:* ${row['Reason']}\n*Sanctioned:* ${row['Sanctioned Year'] || 'N/A'}\n*Cost (Orig/Antic):* ${row['Orig Cost'] || 'N/A'} Cr / ${row['Antic Cost'] || 'N/A'} Cr`;

                const escapedShareText = encodeURIComponent(shareText);

                

                if (searchInput) {

                    const escapedSearchText = searchInput.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

                    const regex = new RegExp(`(${escapedSearchText})`, "gi");

                    projHtml = projHtml.replace(regex, '<mark>$1</mark>');

                }


                tr.innerHTML = `

                    <td style="text-align: center; vertical-align: middle;">

                        <input type="checkbox" class="row-checkbox custom-checkbox" value="${escapedShareText}" onchange="toggleRowSelection(this, 'row-${idx}')">

                    </td>

                    <td style="text-align: center; vertical-align: middle;">

                        <button class="copy-btn" title="Copy for WhatsApp" data-share-text="${escapedShareText}" onclick="copySingle(this)">📋</button>

                    </td>

                    <td><span style="background: #eef2f5; padding: 4px 8px; border-radius: 4px; font-size: 12px; font-weight: bold;">${row['Period']}</span></td>

                    <td style="font-weight: bold; color: #3498db;">${row['Zone']}</td>

                    <td>${projHtml}</td>

                    <td style="font-weight: bold; color: #e67e22;">${row['Reason']}</td>

                    <td>${row['Sanctioned Year']}</td>

                    <td>${row['Length']}</td>

                    <td>${row['Orig Cost']}</td>

                    <td>${row['Antic Cost']}</td>

                `;

                fragment.appendChild(tr);

            });

            

            tbody.appendChild(fragment);

            updateBulkUI();

        }


        // --- Bulk Selection & Copying Logic ---

        

        function toggleRowSelection(checkbox, rowId) {

            const row = document.getElementById(rowId);

            if(checkbox.checked) row.classList.add('selected');

            else row.classList.remove('selected');

            

            checkSelectAllState();

            updateBulkUI();

        }


        function toggleSelectAll() {

            const selectAllCb = document.getElementById('selectAll');

            const checkboxes = document.querySelectorAll('.row-checkbox');

            

            checkboxes.forEach(cb => {

                cb.checked = selectAllCb.checked;

                const row = cb.closest('tr');

                if(cb.checked) row.classList.add('selected');

                else row.classList.remove('selected');

            });

            updateBulkUI();

        }


        function checkSelectAllState() {

            const total = document.querySelectorAll('.row-checkbox').length;

            const checked = document.querySelectorAll('.row-checkbox:checked').length;

            const selectAllCb = document.getElementById('selectAll');

            

            if (total === 0) selectAllCb.checked = false;

            else selectAllCb.checked = (total === checked);

        }


        function updateBulkUI() {

            const checkedBoxes = document.querySelectorAll('.row-checkbox:checked');

            const bulkDiv = document.getElementById('bulkActions');

            const countSpan = document.getElementById('selectedCount');

            

            if(checkedBoxes.length > 0) {

                bulkDiv.style.display = 'flex';

                countSpan.textContent = `${checkedBoxes.length} projects selected`;

            } else {

                bulkDiv.style.display = 'none';

            }

        }


        function copySelected() {

            const checkboxes = document.querySelectorAll('.row-checkbox:checked');

            if (checkboxes.length === 0) return;

            

            let combinedText = "🚧 *Project Delays Summary* 🚧\n\n";

            checkboxes.forEach((cb, index) => {

                // Remove the "Project Delay Alert" header from individual items for bulk copy

                let text = decodeURIComponent(cb.value).replace("🚧 *Project Delay Alert*\n\n", "");

                combinedText += text + "\n";

                if(index < checkboxes.length - 1) {

                    combinedText += "\n➖➖➖➖➖➖➖➖➖➖\n\n";

                }

            });


            navigator.clipboard.writeText(combinedText).then(() => {

                const btn = document.getElementById('bulkCopyBtn');

                const originalText = btn.innerHTML;

                btn.innerHTML = '✅ Copied to Clipboard!';

                btn.classList.add('copied');

                

                document.getElementById('selectAll').checked = false;

                toggleSelectAll();

                

                setTimeout(() => {

                    btn.innerHTML = originalText;

                    btn.classList.remove('copied');

                }, 2500);

            }).catch(err => {

                console.error('Failed to copy text: ', err);

                alert("Failed to copy.");

            });

        }

            }).catch(err => {
